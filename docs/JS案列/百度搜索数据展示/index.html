百度搜索数据展示

逻辑1：先完成 展示区域的显示或隐藏​

onkeyup事件处理程序是在用户释放键盘按键时触发的。
它通常用于检测用户输入的内容，例如，当用户输入用户名或密码时，可以使用onkeyup事件来检查输入的内容是否合法。

onblur事件处理程序是在元素失去焦点时触发的。
它通常用于检测用户是否离开了某个元素，
例如，当用户离开输入框时，可以使用onblur事件来检查输入的内容是否合法。

javascript
<script>
    let input = document.getElementById('val');
    let show = document.getElementById('show');
    input.onkeyup = function () {
        show.style.display = 'block';
    }
    input.onblur = function () {
        show.style.display = 'none';
        input.value = '';
    }
</script>

逻辑2：拿到用户输入的值，去和数据库每一项进行匹配​

show.innerHTML = str 和 show.insertAdjacentHTML('beforeend', str) 这两个方法都是用来添加 HTML 内容到元素中，但是它们的行为和结果是不同的。

show.innerHTML = str

当你使用 innerHTML 属性来设置元素的内容时，浏览器会将新的 HTML 内容完全替换掉元素原有的内容。也就是说，元素的原有内容会被完全清除，然后新的内容会被添加进来。

在你的例子中，当你使用 show.innerHTML = str 时，浏览器会将新的 HTML 内容完全替换掉 show 元素原有的内容。由于 str 是一个包含多个p元素的字符串，所以浏览器会将这些p元素添加到 show 元素中。

show.insertAdjacentHTML('beforeend', str)

当你使用 insertAdjacentHTML 方法来添加 HTML 内容到元素中时，浏览器会将新的 HTML 内容添加到元素的指定位置，而不会清除元素原有的内容。

在你的例子中，当你使用 show.insertAdjacentHTML('beforeend', str) 时，浏览器会将新的 HTML 内容添加到 show 元素的末尾。由于 str 是一个包含多个p元素的字符串，所以浏览器会将这些p元素添加到 show 元素的末尾。

但是，由于你在循环中使用 insertAdjacentHTML 方法，所以每次循环都会将新的 HTML 内容添加到 show 元素的末尾。这导致了 show 元素的内容不断增加，造成了循环的效果。

总之，innerHTML 属性会完全替换掉元素的内容，而 insertAdjacentHTML 方法会将新的内容添加到元素的指定位置。

    let arr=['超神学院 50元','数据结构与算法, 199元','css核心技术,299元','Vue核心技术,399元','笑猫日记,38元']
    let input = document.getElementById('val');
    let show = document.getElementById('show');
    input.onkeyup = function () {
        let str='';
        show.style.display = 'block';
        arr.forEach(item=>{
            let res =item.indexOf(input.value)
            if (res !==-1){
                str+=`<p>${item}</p>`
            }
        })
        if (!input.value || !str){
            show.innerHTML='<p>没有该书籍</p>';
        }else {
            show.innerHTML=str
        }

    }
    input.onblur = function () {
        show.style.display = 'none';
        input.value = '';
    }
</script>

<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>课程搜索页面</title>
    <style>
        /* 页面样式 */
        body {
            font-family: Arial, sans-serif;
            margin: 0;
            padding: 0;
            display: flex;
            justify-content: center;
            align-items: center;

        .content {

        }

        .search {

        .show {

        }

            color: red;
            font-weight: bold;
        }

<body>
<div class="content">
    <div class="search">
        <input type="text" placeholder="请输入课程" id="val">
        <button id="clear">清空</button>
        <button id="submit">提交</button>
    </div>
    <div class="show" id="show"></div>
</div>

<script>
    const searchResults = document.getElementById('show');
    const input = document.getElementById('val');
    const clearBtn = document.getElementById('clear');
    const submitBtn = document.getElementById('submit');
    const recentSearches = [];
    const courses = [
        { name: '超神学院', price: 50 },
        { name: '数据结构与算法', price: 199 },
        { name: 'css核心技术', price: 299 },
        { name: 'Vue核心技术', price: 399 },
        { name: '笑猫日记', price: 38 },
        { name: 'JavaScript高级程序设计', price: 259 },
        { name: '深入理解计算机系统', price: 329 },
        { name: 'Python编程：从入门到实践', price: 199 },
        { name: 'HTML5与CSS3实战', price: 149 },
        { name: 'React核心原理与实践', price: 299 },
        { name: '编程珠玑', price: 179 },
        { name: 'Node.js深入浅出', price: 249 },
        { name: '数据库系统概念', price: 399 },
        { name: '操作系统：设计与实现', price: 349 },
        { name: '算法导论', price: 459 }
    ];

    // 防抖函数
    function debounce(fn, delay) {
        let timer;
        return function () {
            clearTimeout(timer);
            timer = setTimeout(() => fn.apply(this, arguments), delay);
        };
    }

    // 高亮匹配内容
    function highlightText(text, keyword) {
        return text.replace(new RegExp(keyword, "gi"), (match) => `<span class="highlight">${match}</span>`);
    }

    // 搜索课程
    function searchCourses() {
        const searchValue = input.value.trim();
        if (!searchValue) {
            searchResults.innerHTML = '<p>请输入课程名称</p>';
            searchResults.style.display = 'block';
            return;
        }

        const filteredCourses = courses.filter((course) => course.name.includes(searchValue));
        if (filteredCourses.length === 0) {
            searchResults.innerHTML = '<p>没有找到相关课程</p>';
            searchResults.style.display = 'block';
            return;
        }

        const html = filteredCourses.map((course) => {
            return `<p>${highlightText(course.name, searchValue)} - ${course.price}元</p>`;
        }).join('');
        searchResults.innerHTML = html;
        searchResults.style.display = 'block';
    }

    // 回车提交搜索
    input.addEventListener('keyup', debounce((e) => {
        if (e.key === "Enter") {
            submitSearch();
        } else {
            searchCourses();
        }
    }, 300));

    // 清空按钮
    clearBtn.addEventListener('click', () => {
        input.value = '';
        searchResults.style.display = 'none';
    });

    // 提交按钮
    submitBtn.addEventListener('click', submitSearch);

    function submitSearch() {
        const searchValue = input.value.trim();
        if (searchValue) {
            alert("提交的搜索内容: " + searchValue);
            updateRecentSearches(searchValue);
            searchCourses();
        } else {
            alert("请输入课程名称！");
        }
    }

    // 更新最近搜索
    function updateRecentSearches(searchValue) {
        if (!recentSearches.includes(searchValue)) {
            recentSearches.unshift(searchValue);
            if (recentSearches.length > 5) recentSearches.pop(); // 保留最多5条记录
        }
    }

    // 输入框获取焦点时展示最近搜索
    input.addEventListener('focus', () => {
        if (recentSearches.length > 0) {
            const html = recentSearches.map((item) => `<p>${item}</p>`).join('');
            searchResults.innerHTML = `<p><strong>最近搜索：</strong></p>` + html;
            searchResults.style.display = 'block';
        }
    });

    // 输入框失焦隐藏结果
    input.addEventListener('blur', () => {
        setTimeout(() => {
            searchResults.style.display = 'none';
        }, 200); // 避免点击结果时隐藏
    });
</script>
</body>
</html>
